cmake_minimum_required(VERSION 3.26)
project(PBZMods LANGUAGES C CXX ASM_MASM)
if(NOT MSVC OR CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "PBZ requires the MSVC x64 toolchain.")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Game__Shipping__Win64")
    message(FATAL_ERROR "Use the ninja-msvc-x64 preset.")
endif()
# CMake's AUTO decoding can corrupt localized /showIncludes output.
# Ninja matches raw bytes, so preserve the compiler's prefix through generation.
set(pbz_probe_dir "${CMAKE_BINARY_DIR}/CMakeFiles/PBZShowIncludes")
file(MAKE_DIRECTORY "${pbz_probe_dir}")
file(WRITE "${pbz_probe_dir}/pbz_headerdeps_probe.h" "\n")
file(WRITE "${pbz_probe_dir}/probe.c" "#include \"pbz_headerdeps_probe.h\"\n")
foreach(pbz_language IN ITEMS C CXX)
    if(pbz_language STREQUAL "C")
        set(pbz_probe_mode /TC)
    else()
        set(pbz_probe_mode /TP)
    endif()
    execute_process(COMMAND "${CMAKE_${pbz_language}_COMPILER}" /nologo /showIncludes ${pbz_probe_mode} /c probe.c
        WORKING_DIRECTORY "${pbz_probe_dir}" OUTPUT_VARIABLE pbz_probe_output
        ERROR_VARIABLE pbz_probe_error RESULT_VARIABLE pbz_probe_result ENCODING NONE)
    if(NOT pbz_probe_result EQUAL 0 OR NOT pbz_probe_output MATCHES
        "(^|\n)([^\r\n]+)([A-Za-z]:[/\\\\][^\r\n]*[/\\\\]pbz_headerdeps_probe.h)\r?(\n|$)")
        message(FATAL_ERROR "Cannot determine the raw MSVC ${pbz_language} include prefix.")
    endif()
    set(pbz_probe_prefix "${CMAKE_MATCH_2}")
    if(pbz_language STREQUAL "CXX" AND NOT pbz_probe_prefix STREQUAL CMAKE_CL_SHOWINCLUDES_PREFIX)
        message(FATAL_ERROR "C and C++ compilers have different include prefixes.")
    endif()
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${pbz_probe_prefix}")
endforeach()
file(WRITE "${pbz_probe_dir}/prefix.bin" "${CMAKE_CL_SHOWINCLUDES_PREFIX}")

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "Keep populated dependency revisions")
find_program(PBZ_PWSH pwsh REQUIRED)
execute_process(COMMAND "${PBZ_PWSH}" -NoProfile -File "${CMAKE_SOURCE_DIR}/scripts/setup.ps1" -VerifyOnly
    RESULT_VARIABLE pbz_verify_result)
if(NOT pbz_verify_result EQUAL 0)
    message(FATAL_ERROR "Dependency verification failed; run scripts/setup.ps1.")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_SOURCE_DIR}/RE-UE4SS" rev-parse HEAD
    OUTPUT_VARIABLE pbz_source_commit OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE pbz_git_result)
if(NOT pbz_git_result EQUAL 0)
    message(FATAL_ERROR "Cannot read the verified UE4SS revision.")
endif()
if(DEFINED PBZ_CONFIGURED_UE4SS_COMMIT AND NOT PBZ_CONFIGURED_UE4SS_COMMIT STREQUAL pbz_source_commit)
    message(FATAL_ERROR "UE4SS changed; choose a fresh build directory.")
endif()
set(PBZ_CONFIGURED_UE4SS_COMMIT "${pbz_source_commit}" CACHE INTERNAL "Configured UE4SS revision")
enable_testing()
add_subdirectory(RE-UE4SS)
add_subdirectory(mods)
include(cmake/PBZTooling.cmake)
